cmake_minimum_required(VERSION 3.20)
project(vtr VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CTest)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "vtr-cpp currently requires Linux")
endif()
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
find_path(LZ4_INCLUDE_DIR lz4.h REQUIRED)
find_library(LZ4_LIBRARY NAMES lz4 REQUIRED)
find_path(ZSTD_INCLUDE_DIR zstd.h REQUIRED)
find_library(ZSTD_LIBRARY NAMES zstd REQUIRED)
add_library(vtr src/format.cpp src/writer.cpp src/reader.cpp src/c_api.cpp src/c_queries.cpp src/c_activity.cpp src/c_guard.cpp)
add_library(vtr::vtr ALIAS vtr)
target_compile_features(vtr PUBLIC cxx_std_20)
target_include_directories(vtr PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE ${LZ4_INCLUDE_DIR} ${ZSTD_INCLUDE_DIR})
target_include_directories(vtr SYSTEM PRIVATE third_party/unordered_dense/include)
target_link_libraries(vtr PRIVATE Threads::Threads ZLIB::ZLIB ${LZ4_LIBRARY} ${ZSTD_LIBRARY})
set_target_properties(vtr PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(vtr PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS vtr EXPORT vtrTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES third_party/unordered_dense/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/vtr/unordered_dense)
install(EXPORT vtrTargets NAMESPACE vtr:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/vtr)
if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_executable(vtr_c_smoke tests/c_smoke.c)
    target_link_libraries(vtr_c_smoke PRIVATE vtr)
    add_test(NAME c_api COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_c.py $<TARGET_FILE:vtr_c_smoke>)
    add_executable(vtr_interop tests/interop.cpp)
    target_include_directories(vtr_interop PRIVATE src)
    target_link_libraries(vtr_interop PRIVATE vtr)
    add_test(NAME rust_interop COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/interop.py $<TARGET_FILE:vtr_interop>)
    file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/track)
    add_executable(vtr_track_test tests/track_test.cpp)
    target_include_directories(vtr_track_test PRIVATE tests/dpi_stub)
    target_link_libraries(vtr_track_test PRIVATE vtr)
    add_test(NAME simulator_tracking COMMAND vtr_track_test ${CMAKE_CURRENT_BINARY_DIR}/track)
    add_executable(vtr_native_test tests/native.cpp)
    target_include_directories(vtr_native_test PRIVATE src)
    target_link_libraries(vtr_native_test PRIVATE vtr)
    add_test(NAME native_invariants COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_native.py $<TARGET_FILE:vtr_native_test>)
    add_executable(vtr_guard_test tests/guard.cpp)
    target_link_libraries(vtr_guard_test PRIVATE vtr)
    add_test(NAME crash_and_stop COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/guard.py $<TARGET_FILE:vtr_guard_test>)
endif()
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/vtrConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/vtrConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/vtr)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/vtrConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/vtrConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/vtrConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/vtr)
configure_file(cmake/vtr.pc.in ${CMAKE_CURRENT_BINARY_DIR}/vtr.pc @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/vtr.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
