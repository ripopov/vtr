cmake_minimum_required(VERSION 3.16)
project(vtr_logging_demo CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../..)
find_package(Threads REQUIRED)
# Build the library first: cargo build --release (produces target/release/libvtr.a)
add_executable(sim_log_demo sim_log_demo.cpp)
target_include_directories(sim_log_demo PRIVATE ${ROOT}/crates/vtr-capi/include)
target_link_libraries(sim_log_demo PRIVATE ${ROOT}/target/release/libvtr.a Threads::Threads ${CMAKE_DL_LIBS} m)
target_compile_options(sim_log_demo PRIVATE -O2)
